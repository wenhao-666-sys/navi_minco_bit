cmake_minimum_required(VERSION 3.8)
project(nav_serial_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/SerialSendData.msg" "msg/SerialReceiveData.msg" DEPENDENCIES std_msgs)
add_library(nav_uart src/qyg_protocol.cpp src/uart_transporter.cpp)
target_include_directories(nav_uart PUBLIC include)
add_executable(nav_serial_node src/nav_serial_node.cpp)
target_link_libraries(nav_serial_node nav_uart)
ament_target_dependencies(nav_serial_node rclcpp geometry_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(nav_serial_node "${cpp_typesupport_target}")
install(TARGETS nav_serial_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/foxglove_velocity.bash DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_protocol test/test_protocol.cpp)
  target_link_libraries(test_protocol nav_uart)
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
